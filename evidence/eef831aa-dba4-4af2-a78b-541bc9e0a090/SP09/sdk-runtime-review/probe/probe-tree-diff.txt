+//! Direct checks of production scrub and header construction.
+//! Assertions report lengths and marker classes. They do not print secret text.
+
+use reqwest::header::{AUTHORIZATION, HeaderMap};

+use crate::token::BearerToken;
+use crate::wire::scrub_text;

+const ENDPOINT: &str = "http://127.0.0.1:9/mount/v1/query";
+
+fn scrub(text: &str, token: &str, sql: &str) -> String {
+    scrub_text(text, token, sql, ENDPOINT)

+fn absent(hay: &str, needle: &str) -> bool {
+    !needle.is_empty() && !hay.contains(needle)
+}

+#[test]
+fn a090_sensitive_header_value() {
+    let mut redacted_map = String::new();
+    for token in ["!", "Z9b8c7d6", "a"] {
+        let bearer = BearerToken::new(token).expect("accepted length");
+        let header = bearer.authorization();
+        let header_debug = format!("{header:?}");
+        assert!(
+            header.is_sensitive() && header_debug == "Sensitive",
+            "sensitive {} debug_len {}",
+            u8::from(header.is_sensitive()),
+            header_debug.len()
+        );
+        let mut map = HeaderMap::new();
+        map.insert(AUTHORIZATION, header);
+        let map_debug = format!("{map:?}");
+        assert!(
+            map_debug.contains("Sensitive"),
+            "map_len {}",
+            map_debug.len()
+        );
+        if redacted_map.is_empty() {
+            redacted_map = map_debug.clone();
+        } else {
+            assert_eq!(map_debug, redacted_map, "map_len {}", map_debug.len());
+        }
+        if token.len() >= 4 {
+            assert!(absent(&map_debug, token), "map_len {}", map_debug.len());
+        }
+        let shown = format!("{bearer:?} {bearer}");
+        assert_eq!(
+            shown,
+            "BearerToken([redacted]) BearerToken([redacted])",
+            "shown_len {}",
+            shown.len()
+        );
+        if token.len() >= 4 {
+            assert!(absent(&shown, token), "shown_len {}", shown.len());
+        }
+        assert_eq!(bearer.secret().len(), token.len());
+    }
+    assert!(BearerToken::new("").is_err(), "empty rejected");
+    assert!(BearerToken::new("a b").is_err(), "space rejected");
+    assert!(BearerToken::new(&"a".repeat(4096)).is_ok(), "4096 accepted");
+    assert!(BearerToken::new(&"a".repeat(4097)).is_err(), "4097 rejected");
+    assert!(oo_core::text::QuerySql::new("x").is_ok(), "sql len 1");
+    assert!(oo_core::text::QuerySql::new("a\nb雪").is_ok(), "newline unicode");
+    assert!(oo_core::text::QuerySql::new("a\0b").is_err(), "nul rejected");
+}

+#[derive(Clone, Copy)]
+enum Retry {
+    None,
+    Admission,
+}
+
+#[derive(Clone, Copy)]
+enum After {
+    Skip,
+    None,
+    Some(u128),
+}
+
+struct Spec {
+    id: &'static str,
+    required: bool,
+    token: String,
+    sql: String,
+    retry: Retry,
+    overall_ms: Option<u64>,
+    kind: &'static str,
+    hits: Option<u32>,
+    elapsed: Option<(u128, u128)>,
+    retry_after: After,
+    body_auth_path: bool,
+    message_lacks_token: bool,
+    message_lacks_sql: bool,
+    hint_lacks_token: bool,
+    hint_eq: Option<&'static str>,
+    message_prefix: Option<&'static str>,
+    lacks: &'static [&'static str],
+    nested_left: bool,
+    literal_u_left: bool,
+    build: Arc<dyn Fn(&Ctx) -> Vec<Step> + Send + Sync>,
+}
+
+struct Ctx {
+    full: String,
+    origin: String,
+    authority: String,
+    path: String,
+    mount: String,
+}
+
+#[derive(Clone)]
+struct Step {
+    bytes: Vec<u8>,
+    cut: Option<usize>,
+}
+
+#[derive(Clone)]
+struct Hit {
+    auth_exact: bool,
+    body_exact: bool,
+    path_exact: bool,
+    auth_count: usize,
+    mismatch: Option<usize>,
+}
+
+async fn run(spec: Spec) -> Value {
+    let listener = TcpListener::bind("127.0.0.1:0").expect("bind");
+    let port = listener.local_addr().expect("port").port();
+    let ctx = Ctx {
+        full: format!("http://127.0.0.1:{port}/mount/v1/query"),
+        origin: format!("http://127.0.0.1:{port}"),
+        authority: format!("127.0.0.1:{port}"),
+        path: "/mount/v1/query".to_owned(),
+        mount: "/mount".to_owned(),
+    };
+    let steps = (spec.build)(&ctx);
+    let expected_body = wire_text(&spec.sql).into_bytes();
+    let expected_auth = format!("Bearer {}", spec.token).into_bytes();
+    let hits = Arc::new(Mutex::new(Vec::<Hit>::new()));
+    let done = Arc::new(AtomicBool::new(false));
+    let ready = Arc::new(AtomicBool::new(false));
+    let seen = Arc::new(AtomicUsize::new(0));
+    let server_hits = Arc::clone(&hits);
+    let server_done = Arc::clone(&done);
+    let server_ready = Arc::clone(&ready);
+    let server_seen = Arc::clone(&seen);
+    let server_steps = steps.clone();
+    let server_body = expected_body.clone();
+    let server_auth = expected_auth.clone();
+    let joined = thread::spawn(move || {
+        listener.set_nonblocking(true).expect("nonblocking");
+        server_ready.store(true, Ordering::SeqCst);
+        while !server_done.load(Ordering::SeqCst) {
+            match listener.accept() {
+                Ok((sock, _)) => {
+                    let index = server_seen.fetch_add(1, Ordering::SeqCst);
+                    let step = server_steps.get(index).cloned().unwrap_or_else(|| empty_500());
+                    record(sock, &step, &server_hits, &server_auth, &server_body, "/mount/v1/query");
+                    if index >= 8 {
+                        break;
+                    }
+                }
+                Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
+                    thread::sleep(Duration::from_millis(5));
+                }
+                Err(_) => break,
+            }
+        }
+    });
+    while !ready.load(Ordering::SeqCst) {
+        thread::sleep(Duration::from_millis(1));
+    }
+
+    let token = BearerToken::new(&spec.token).expect("token accepted");
+    let mut config = ClientConfig::local_development().with_retry(match spec.retry {
+        Retry::None => RetryPolicy::none(),
+        Retry::Admission => RetryPolicy::admission(3, Duration::from_secs(2)).expect("policy"),
+    });
+    if let Some(overall) = spec.overall_ms {
+        config = config.with_timeouts(timeouts(overall)).expect("timeouts");
+    }
+    let base = format!("http://127.0.0.1:{port}/mount");
+    let client = Client::new(&base, token, config).expect("client");
+    let client_debug = format!("{client:?}");
+    let started = Instant::now();
+    let result = client.query(request(&spec.sql)).await;
+    let elapsed = started.elapsed().as_millis();
+    done.store(true, Ordering::SeqCst);
+    let _ = joined.join();
+    let hit_list = hits.lock().expect("hits").clone();
+
+    let mut kind = String::from("ok");
+    let mut message = String::new();
+    let mut hint = String::new();
+    let mut display = String::new();
+    let mut debug = String::new();
+    let mut source_none = true;
+    let mut retry_after_ms = None;
+    if let Err(error) = &result {
+        kind = class(error);
+        display = format!("{error}");
+        debug = format!("{error:?}");
+        source_none = error.source().is_none();
+        retry_after_ms = error.retry_after().map(|wait| wait.as_millis());
+        if let Some(gateway) = error.gateway() {
+            message = gateway.message().to_owned();
+            hint = gateway.hint().unwrap_or("").to_owned();
+        }
+    }
+
+    let auth_exact = !hit_list.is_empty() && hit_list.iter().all(|hit| hit.auth_exact && hit.auth_count == 1);
+    let body_exact = !hit_list.is_empty() && hit_list.iter().all(|hit| hit.body_exact);
+    let path_exact = !hit_list.is_empty() && hit_list.iter().all(|hit| hit.path_exact);
+    let mismatch = hit_list.iter().find_map(|hit| hit.mismatch);
+    let hit_count = hit_list.len() as u32;
+
+    let mut ok = kind == spec.kind;
+    if let Some(expected_hits) = spec.hits {
+        ok &= hit_count == expected_hits;
+    }
+    if let Some((min, max)) = spec.elapsed {
+        ok &= elapsed >= min && elapsed <= max;
+    }
+    if spec.body_auth_path {
+        ok &= auth_exact && body_exact && path_exact;
+    }
+    if spec.kind != "ok" {
+        if spec.message_lacks_token {
+            ok &= !message.contains(&spec.token);
+        }
+        if spec.message_lacks_sql {
+            ok &= !message.contains(&spec.sql);
+        }
+        if spec.hint_lacks_token {
+            ok &= !hint.contains(&spec.token);
+        }
+        if let Some(expected) = spec.hint_eq {
+            ok &= hint == expected;
+        }
+        if let Some(prefix) = spec.message_prefix {
+            ok &= message.starts_with(prefix);
+        }
+        for piece in spec.lacks {
+            ok &= !message.contains(piece) && !hint.contains(piece);
+        }
+        ok &= source_none;
+        if spec.token.len() >= 4 {
+            ok &= !display.contains(&spec.token) && !debug.contains(&spec.token);
+        }
+        if spec.sql.len() >= 2 {
+            ok &= !display.contains(&spec.sql) && !debug.contains(&spec.sql);
+        }
+        ok &= !display.contains("127.0.0.1") && !debug.contains("127.0.0.1");
+        ok &= !display.contains("/mount") && !debug.contains("/mount");
+        if spec.kind == "gateway" {
+            ok &= display.contains("q_abc123") && debug.contains("q_abc123");
+        }
+    }
+    if spec.token.len() >= 4 {
+        ok &= !client_debug.contains(&spec.token);
+    }
+    if spec.sql.len() >= 2 {
+        ok &= !client_debug.contains(&spec.sql);
+    }
+    ok &= !client_debug.contains("127.0.0.1");
+    ok &= !client_debug.contains("/mount");
+    let nested_left = message.contains("%2541");
+    if spec.nested_left {
+        ok &= nested_left && !message.contains(&spec.token);
+    }
+    let literal_probe = format!("\\u{:04x}", spec.token.as_bytes().first().copied().unwrap_or(0));
+    let literal_u_left = message.contains(&literal_probe);
+    if spec.literal_u_left {
+        ok &= literal_u_left && !message.contains(&spec.token);
+    }
+    match spec.retry_after {
+        After::Skip => {}
+        After::None => ok &= retry_after_ms.is_none(),
+        After::Some(ms) => ok &= retry_after_ms == Some(ms),
+    }
+    let marker_hit = ["[redacted]", "[omitted]", "****", ""].contains(&message.as_str());
+    if spec.id.starts_with("marker_") {
+        ok &= marker_hit;
+    }
+    if spec.id == "static_message_preserved" {
+        ok &= message == "slow down";
+    }
+
+    json!({
+        "id": spec.id,
+        "required": spec.required,
+        "contract_ok": ok,
+        "hits": hit_count,
+        "elapsed_ms": elapsed,
+        "kind": kind,
+        "message_len": message.len(),
+        "hint_len": hint.len(),
+        "message_has_token": message.contains(&spec.token),
+        "message_has_sql": message.contains(&spec.sql),
+        "hint_has_token": hint.contains(&spec.token),
+        "display_has_token": display.contains(&spec.token),
+        "debug_has_token": debug.contains(&spec.token),
+        "client_debug_has_token": client_debug.contains(&spec.token),
+        "client_debug_has_locator": client_debug.contains("127.0.0.1") || client_debug.contains("/mount"),
+        "source_none": source_none,
+        "retry_after_ms": retry_after_ms,
+        "auth_exact": auth_exact,
+        "body_exact": body_exact,
+        "path_exact": path_exact,
+        "body_mismatch_at": mismatch,
+        "nested_left": nested_left,
+        "literal_u_left": literal_u_left,
+        "message_is_marker": marker_hit,
+        "display_has_query_id": display.contains("q_abc123")
+    })
+}

+fn wire_text(sql: &str) -> String {

+fn json_string(value: &str) -> String {
+    let mut out = String::from('"');
+    for ch in value.chars() {
+        match ch {
+            '"' => out.push_str("\\\""),
+            '\\' => out.push_str("\\\\"),
+            '\n' => out.push_str("\\n"),
+            '\r' => out.push_str("\\r"),
+            '\t' => out.push_str("\\t"),
+            other if (other as u32) < 0x20 => out.push_str(&format!("\\u{:04x}", other as u32)),
+            other => out.push(other),
+        }
+    }
+    out.push('"');
+    out
+}

+fn gateway(
+    status: u16,
+    code: &str,
+    retryable: bool,
+    message: &str,
+    hint: &str,
+    extra: &[&str],
+    _marker: &str,
+) -> Step {
+    let body = serde_json::json!({
+        "error": {
+            "code": code,
+            "message": message,
+            "hint": hint,
+            "retryable": retryable,
+            "query_id": "q_abc123"
+        }

+fn http_step(status: u16, extra: &[&str], body: &[u8], cut: Option<usize>) -> Step {
+    let mut head = format!(
+        "HTTP/1.1 {status} status\r\ncontent-type: application/json\r\ncontent-length: {}\r\nconnection: close\r\n",
+        body.len()
+    );
+    for header in extra {
+        head.push_str(header);
+        head.push_str("\r\n");
+    }
+    head.push_str("\r\n");
+    let mut bytes = head.into_bytes();
+    bytes.extend_from_slice(body);
+    Step { bytes, cut }
+}

+fn percent_encode(text: &str, lower: bool) -> String {
+    let hex = if lower { b"0123456789abcdef" } else { b"0123456789ABCDEF" };
+    let mut out = String::new();
+    for byte in text.bytes() {
+        out.push('%');
+        out.push(hex[(byte >> 4) as usize] as char);
+        out.push(hex[(byte & 0x0f) as usize] as char);
+    }
+    out
+}

+fn unicode_escapes(text: &str) -> String {
+    let mut out = String::new();
+    for byte in text.bytes() {
+        out.push_str(&format!("\\u{byte:04x}"));
+    }
+    out
+}

+fn record(mut sock: TcpStream, step: &Step, hits: &Mutex<Vec<Hit>>, auth: &[u8], body: &[u8], path: &str) {
+    let _ = sock.set_read_timeout(Some(Duration::from_secs(2)));
+    let _ = sock.set_write_timeout(Some(Duration::from_secs(2)));
+    let mut buf = Vec::new();
+    let mut tmp = [0u8; 4096];
+    loop {
+        if buf.windows(4).any(|window| window == b"\r\n\r\n") || buf.len() > 65_536 {
+            break;
+        }
+        match sock.read(&mut tmp) {
+            Ok(0) | Err(_) => break,
+            Ok(count) => buf.extend_from_slice(&tmp[..count]),
+        }
+    }
+    let header_end = buf.windows(4).position(|window| window == b"\r\n\r\n").unwrap_or(buf.len());
+    let head = String::from_utf8_lossy(&buf[..header_end.min(buf.len())]).to_string();
+    let mut content_length = 0usize;
+    let mut auth_values = Vec::new();
+    let mut request_path = String::new();
+    for (index, line) in head.split("\r\n").enumerate() {
+        if index == 0 {
+            request_path = line.split(' ').nth(1).unwrap_or("").to_owned();
+        } else if let Some((name, value)) = line.split_once(':') {
+            if name.eq_ignore_ascii_case("content-length") {
+                content_length = value.trim().parse().unwrap_or(0);
+            }
+            if name.eq_ignore_ascii_case("authorization") {
+                auth_values.push(value.trim().as_bytes().to_vec());
+            }
+        }
+    }
+    let already = buf.len().saturating_sub(header_end.saturating_add(4));
+    let mut got = buf.get(header_end.saturating_add(4)..).unwrap_or(&[]).to_vec();
+    while got.len() < content_length && got.len() < 1_048_576 {
+        match sock.read(&mut tmp) {
+            Ok(0) | Err(_) => break,
+            Ok(count) => got.extend_from_slice(&tmp[..count]),
+        }
+    }
+    let _ = already;
+    if got.len() > content_length {
+        got.truncate(content_length);
+    }
+    let mismatch = if got == body {
+        None
+    } else {
+        Some(
+            got.iter()
+                .zip(body.iter())
+                .position(|(left, right)| left != right)
+                .unwrap_or(got.len().min(body.len())),
+        )
+    };
+    hits.lock().expect("record").push(Hit {
+        auth_exact: auth_values.len() == 1 && auth_values[0] == auth,
+        body_exact: got == body,
+        path_exact: request_path == path,
+        auth_count: auth_values.len(),
+        mismatch,
+    });
+    let write_len = step.cut.unwrap_or(step.bytes.len()).min(step.bytes.len());
+    let _ = sock.write_all(&step.bytes[..write_len]);
+    let _ = sock.flush();
+}
